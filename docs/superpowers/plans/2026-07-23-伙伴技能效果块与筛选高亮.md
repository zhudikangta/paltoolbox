> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

核心导出以下函数：

```js
function unique(values) {
    return (values || []).filter(function(value, index, all) {
        return value && all.indexOf(value) === index;
    });
}

function blockCapabilityIds(block) {
    return unique((block.subcategoryIds || []).concat(block.tagIds || []));
}

function applyEffectBlocks(options) {
    const partnerSkills = JSON.parse(JSON.stringify(options.partnerSkills || {}));
    const catalog = options.catalog || [];
    const taxonomy = options.taxonomy || { groups: [], detailTags: [] };
    const definitions = options.definitions && options.definitions.partnerSkills || {};
    const knownSubcategories = new Set();
    const knownTags = new Set();
    const errors = [];

    (taxonomy.groups || []).forEach(function(group) {
        (group.children || []).forEach(function(child) { knownSubcategories.add(child.id); });
    });
    (taxonomy.detailTags || []).forEach(function(tag) { knownTags.add(tag.id); });

    catalog.forEach(function(item) {
        const blocks = definitions[item.palId];
        if (!Array.isArray(blocks) || !blocks.length) {
            errors.push('效果块未覆盖目录: ' + item.palId);
            return;
        }
        blocks.forEach(function(block, index) {
            if (!block || !String(block.text || '').trim()) {
                errors.push(item.palId + ' 第 ' + (index + 1) + ' 个效果块正文为空');
            }
            (block.subcategoryIds || []).forEach(function(id) {
                if (!knownSubcategories.has(id)) errors.push(item.palId + ' 引用未知下级分类: ' + id);
            });
            (block.tagIds || []).forEach(function(id) {
                if (!knownTags.has(id)) errors.push(item.palId + ' 引用未知精确标签: ' + id);
            });
        });

        const blockSubcategories = unique([].concat.apply([], blocks.map(function(block) {
            return block.subcategoryIds || [];
        }))).sort();
        const blockTags = unique([].concat.apply([], blocks.map(function(block) {
            return block.tagIds || [];
        }))).sort();
        const catalogSubcategories = unique(item.usageSubcategoryIds || []).sort();
        const catalogTags = unique(item.usageTagIds || []).sort();

        if (JSON.stringify(blockSubcategories) !== JSON.stringify(catalogSubcategories)) {
            errors.push(item.palId + ' 的效果块下级分类与现有分类不一致');
        }
        if (JSON.stringify(blockTags) !== JSON.stringify(catalogTags)) {
            errors.push(item.palId + ' 的效果块精确标签与现有分类不一致');
        }

        partnerSkills[item.palId].effectBlocks = JSON.parse(JSON.stringify(blocks));
        partnerSkills[item.palId].description = blocks.map(function(block) {
            return String(block.text || '').trim();
        }).join('\n');
    });

    if (errors.length) throw new Error(errors.join('\n'));
    return { partnerSkills, catalog };
}

module.exports = { unique, blockCapabilityIds, applyEffectBlocks };
```

```js
function selectedPartnerOptionIds(facetSelections) {
    return Object.keys(facetSelections || {}).reduce(function(ids, facetId) {
        return ids.concat(Array.isArray(facetSelections[facetId]) ? facetSelections[facetId] : []);
    }, []);
}

function getPartnerEffectBlockModels(item, facetSelections) {
    const selectedIds = selectedPartnerOptionIds(facetSelections);
    return (item && item.effectBlocks || []).map(function(block) {
        const capabilityIds = (block.subcategoryIds || []).concat(block.tagIds || []);
        const labels = [];

        (block.subcategoryIds || []).forEach(function(id) {
            const definition = partnerSubcategoryById[id];
            if (!definition || definition.filterable === false) return;
            labels.push({ id, label: definition.label || id, selected: selectedIds.indexOf(id) > -1 });
        });
        (block.tagIds || []).forEach(function(id) {
            const definition = partnerDetailTagById[id];
            if (!definition || definition.kind !== 'precise') return;
            labels.push({ id, label: definition.label || id, selected: selectedIds.indexOf(id) > -1 });
        });

        const highlighted = selectedIds.some(function(optionId) {
            const option = partnerFacetOptionById[optionId];
            const requiredIds = option ? option.capabilityIds : [optionId];
            return requiredIds.some(function(id) { return capabilityIds.indexOf(id) > -1; });
        });

        return {
            text: block.text || '',
            labels: labels,
            highlighted: highlighted
        };
    });
}
```
